Graphing

ggplot2

Basics

ggplot() + geom_point()

use case: scatter plot of two continuous variables

ggplot(df, aes(x = xvar, y = yvar)) +
  geom_point()

geom_bar()

use case: bar chart of counts by category (ggplot does the counting)

ggplot(df, aes(x = categorical_var)) +
  geom_bar()

geom_col()

use case: bar chart from values you've already summarized

ggplot(df, aes(x = categorical_var, y = precomputed_value)) +
  geom_col()

geom_line()

use case: line plot, e.g. a trend over time

ggplot(df, aes(x = xvar, y = yvar)) +
  geom_line()

geom_histogram()

use case: distribution of a single continuous variable

ggplot(df, aes(x = xvar)) +
  geom_histogram(binwidth = 5)

geom_density()

use case: smoothed distribution, good for comparing shapes across groups

ggplot(df, aes(x = xvar, fill = groupvar)) +
  geom_density(alpha = 0.5)

geom_boxplot()

use case: compare a distribution across groups

ggplot(df, aes(x = groupvar, y = yvar)) +
  geom_boxplot()

geom_smooth()

use case: overlay a trend/regression line, e.g. on top of geom_point()

ggplot(df, aes(x = xvar, y = yvar)) +
  geom_point() +
  geom_smooth(method = "lm")  # "loess" for a curved local trend
Customization

labs()

use case: set the title, axis labels, and legend title

+ labs(
    title = "Plot Title",
    x = "X Axis",
    y = "Y Axis",
    color = "Legend Title"
  )

theme()

use case: adjust non-data plot elements — fonts, gridlines, legend position

+ theme_minimal() +
  theme(legend.position = "bottom")

facet_wrap() / facet_grid()

use case: split into small multiples by a grouping variable

+ facet_wrap(~ groupvar)

# two grouping variables, one per row/column
+ facet_grid(rowvar ~ colvar)

scale_*_manual()

use case: manually set colors/fills for discrete groups

+ scale_color_manual(values = c(
    "group1" = "#1a2b57",
    "group2" = "#d63384"
  ))

coord_flip()

use case: flip x/y axes, e.g. to make long category labels readable

ggplot(df, aes(x = categorical_var, y = yvar)) +
  geom_col() +
  coord_flip()

xlim() / ylim()

use case: manually set axis limits

+ xlim(0, 100) +
  ylim(0, 1)
Text, Labels & Saving

geom_text() / geom_label()

use case: annotate points with data labels (geom_label adds a background box)

ggplot(df, aes(x = xvar, y = yvar, label = idvar)) +
  geom_point() +
  geom_text(vjust = -1)

ggsave()

use case: save the last plot (or a stored one) to a file

ggsave("plot.png", width = 8, height = 5, dpi = 300)

# save a specific plot object instead of the last one drawn
ggsave("plot.png", plot = my_plot)
