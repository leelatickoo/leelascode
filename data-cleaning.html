mutate()

use case: Create, modify, and delete columns

%>%
  # +, -, log(), etc. — usual math
  mutate(total = colname1 + colname2)
  mutate(difference = colname1 - colname2)
  mutate(log_colname = log(colname))
  # lead(), lag()
  mutate(next_value = lead(colname))
  mutate(previous_value = lag(colname))
  # dense_rank(), min_rank(), percent_rank(), row_number(), cume_dist(), ntile()
  mutate(
    dense_rank_col   = dense_rank(colname),
    min_rank_col     = min_rank(colname),
    percent_rank_col = percent_rank(colname),
    row_num_col      = row_number(colname),
    cume_dist_col    = cume_dist(colname),
    ntile_col        = ntile(colname, 4)
  )
  # cumsum(), cummean(), cummin(), cummax(), cumany(), cumall()
  mutate(
    running_total = cumsum(colname),
    running_mean  = cummean(colname),
    running_min   = cummin(colname),
    running_max   = cummax(colname),
    any_so_far    = cumany(colname > 0),
    all_so_far    = cumall(colname > 0)
  )
  # na_if(), coalesce()
  mutate(cleaned_col = na_if(colname, -999))
  mutate(combined_col = coalesce(colname1, colname2, colname3))
  # if_else(), recode(), case_when()
  mutate(flag = if_else(colname > 50, "high", "low"))
  mutate(recoded_col = recode(colname,
    "old_value1" = "new_value1",
    "old_value2" = "new_value2"
  ))
  mutate(category = case_when(
    colname < 0  ~ "negative",
    colname == 0 ~ "zero",
    colname > 0  ~ "positive",
    TRUE ~ NA_character_
  ))

summarize()

use case: Summarise each group down to one row description: collapses your data down to one row per group (or one row total if ungrouped)

%>%
  group_by(groupcol) %>%
  summarize(avg_value = mean(colname, na.rm = TRUE))

%>%
  group_by(groupcol) %>%
  summarize(
    avg_value = mean(colname, na.rm = TRUE),
    total     = sum(colname, na.rm = TRUE),
    count     = n(),
    std_dev   = sd(colname, na.rm = TRUE)
  )
  # grouping by multiple columns
  group_by(groupcol1, groupcol2) %>%
  summarize(avg_value = mean(colname, na.rm = TRUE))
  # no grouping — collapses the whole dataframe to 1 row
  summarize(avg_value = mean(colname, na.rm = TRUE))

reframe()

use case: Transform each group to an arbitrary number of rows description: does the same kind of thing, but is built for when each group needs to return a different number of rows (or more than one row)

%>%
  # quantiles: 3 rows returned per group
  group_by(groupcol) %>%
  reframe(quantile_value = quantile(colname, probs = c(0.25, 0.5, 0.75)))
  # range: 2 rows returned per group (min, max)
  group_by(groupcol) %>%
  reframe(range_value = range(colname, na.rm = TRUE))
  # unique(): a different number of rows per group, depending on the data
  group_by(groupcol) %>%
  reframe(unique_value = unique(colname))

Mutating Joins

use case: combine columns from two tables

# inner_join() — keep only rows that match in both
inner_join(dataframe1, dataframe2, by = "key_col")
# left_join() — keep all rows from dataframe1, add matches from dataframe2
left_join(dataframe1, dataframe2, by = "key_col")
# right_join() — keep all rows from dataframe2, add matches from dataframe1
right_join(dataframe1, dataframe2, by = "key_col")
# full_join() — keep all rows from both, matched where possible
full_join(dataframe1, dataframe2, by = "key_col")


# When the key columns have different names in each table
left_join(dataframe1, dataframe2, by = c("key_col1" = "key_col2"))
# Joining on multiple key columns at once
left_join(dataframe1, dataframe2, by = c("key_col1", "key_col2"))
# Handling overlapping column names (non-key columns shared by both tables)
left_join(dataframe1, dataframe2, by = "key_col", suffix = c("_1", "_2"))
# Declaring the expected match type — warns/errors if violated, good for catching duplicate-key bugs
left_join(dataframe1, dataframe2, by = "key_col", relationship = "many-to-one")

Filtering Joins

use case: keep/drop rows from dataframe1, never add columns

# semi_join() — keep rows in dataframe1 that DO have a match in dataframe2
semi_join(dataframe1, dataframe2, by = "key_col")

# anti_join() — keep rows in dataframe1 that have NO match in dataframe2
anti_join(dataframe1, dataframe2, by = "key_col")

A practical distinction worth remembering: mutating joins (inner_join, left_join, etc.) can add columns from the second table; filtering joins (semi_join, anti_join) only ever filter the rows of the first table and never bring in new columns useful when you just want to check existence, not pull in data

Set operations

distinction: both tables need identical column names/types

union(dataframe1, dataframe2)        # combine, drop duplicate rows
union_all(dataframe1, dataframe2)    # combine, keep duplicate rows
intersect(dataframe1, dataframe2)    # rows that appear in both
setdiff(dataframe1, dataframe2)      # rows in dataframe1 not in dataframe2
